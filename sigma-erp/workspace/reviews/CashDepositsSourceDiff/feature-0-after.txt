import { CanDeactivateFn, Routes } from '@angular/router';
import { CashDepositsComponent } from './components/list/list.component';
import { DetailsComponent } from './components/details/details.component';

const preventUnsavedExit: CanDeactivateFn<DetailsComponent> = (component) =>
  component.canDeactivate();

export const CashDeposit_Routes: Routes = [
  { path: 'create', component: DetailsComponent, data: { mode: 'create' }, canDeactivate: [preventUnsavedExit] },
  { path: 'view/:id', component: DetailsComponent, data: { mode: 'view' }, canDeactivate: [preventUnsavedExit] },
  { path: 'edit/:id', component: DetailsComponent, data: { mode: 'edit' }, canDeactivate: [preventUnsavedExit] },
  { path: 'details/:id', redirectTo: 'view/:id' },
  { path: '', component: CashDepositsComponent },
];
