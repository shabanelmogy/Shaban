using SiGma.Helpers.ViewModelsBases.Update;
using SiGma.ViewModels.ViewModels.CashDepositDetail;
using System.ComponentModel.DataAnnotations;

namespace SiGma.ViewModels.ViewModels.CashDeposit
{
    public class CashDepositUpdateVM : UpdateBaseVm
    {
        [Required] public DateTime Date { get; set; }

        [Required, Range(1, int.MaxValue)] public int BankId { get; set; }

        [MaxLength(100)] public string? RefNo { get; set; }

        [MaxLength(500)] public string? Description { get; set; }

        [Required, Range(1, int.MaxValue)] public int PaymentMethodId { get; set; }

        public int? BranchId { get; set; }

        /// <summary>Detail lines — each must include their own Id for differential tracking.</summary>
        [Required, MinLength(1)] public List<CashDepositDetailUpdateVM> CashDepositDetails { get; set; } = [];
    }
}
