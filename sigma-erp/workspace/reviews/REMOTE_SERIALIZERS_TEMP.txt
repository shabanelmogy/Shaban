rememberListState(query: BillListQuery): void {
    const owner = this.listStateOwner();
    this.lastListState = owner ? { owner, query: { ...query, filters: { ...query.filters } } } : null;
  }
readListState(): BillListQuery | null {
    if (!this.lastListState || this.lastListState.owner !== this.listStateOwner()) {
      this.lastListState = null;
      return null;
    }
    return { ...this.lastListState.query, filters: { ...this.lastListState.query.filters } };
  }

search(request: VehicleSearchModel, options?: { skipErrorInterceptor?: boolean }): Observable<Results<VehicleListModel>> {
    let params = new HttpParams();
    Object.entries(request).forEach(([key, value]) => {
      if (value === undefined || value === '') return;
      if (value === null) {
        // Vehicle/Search treats an empty pageSize as the confirmed unpaged contract.
        if (key === 'pageSize') params = params.set(key, '');
        return;
      }

      params = params.set(key, String(value));
    });
    return this.http.get<Results<VehicleListModel>>(`${environment.baseUrl}Vehicle/Search`, {
      params, ...(options?.skipErrorInterceptor ? { headers: { 'X-Skip-Error-Interceptor': 'true' } } : {}),
    });
  }
