  readonly pageNumber = signal(1);

  readonly errorMessage = signal('');
  readonly lookupError = signal(false);
  readonly loadingLookups = signal(false);
  readonly exporting = signal(false);
  readonly reporting = signal(false);
  readonly actionBusy = signal(false);
  readonly branchOptions = signal<DropDownSelect[]>([]);
  readonly customerOptions = signal<DropDownSelect[]>([]);
  readonly vehicleOptions = signal<DropDownSelect[]>([]);
  readonly advancedFilterNames = ['vehicleId', 'branchId', 'chassisNo'];
  readonly statusOptions = [{ id: StatusEnum.Open, key: 'workOrder.open' }, { id: StatusEnum.Closed, key: 'workOrder.closed' }];
  readonly acceptanceStatusOptions = [

    vehicleId: this.fb.control<number | null>(null),
    customerId: this.fb.control<number | null>(null),
    branchId: this.fb.control<number | null>(null),
    chassisNo: this.fb.nonNullable.control(''),
  });
  @ViewChild('noTemplate', { static: true }) protected noTemplate!: TemplateRef<DataTableCellContext<WorkOrderListModel>>;
  columns: ReadonlyArray<DataTableColumn<WorkOrderListModel>> = [];
  readonly moreActions: ActionList<WorkOrderListModel>[] = [

      action: (row) => { if (row) this.confirmAction(row, 'delete'); } },

  constructor() {

  ngOnInit(): void {

      { field: 'customerNo', header: 'workOrder.customerNo' },
      { field: 'branch', header: 'workOrder.branch' },
      { field: 'vehicle', header: 'workOrder.vehicle' },
      { field: 'totalAmount', header: 'workOrder.totalAmount', type: 'money' },
      { field: 'discount', header: 'workOrder.discount', type: 'money' },
      { field: 'netAmount', header: 'workOrder.netAmount', type: 'money' },
      { field: 'status', header: 'workOrder.status', type: 'status', status: (row) => ({
        label: Number(row.status) === StatusEnum.Open ? 'workOrder.open' : 'workOrder.closed',
        tone: Number(row.status) === StatusEnum.Open ? 'success' : 'neutral',
      }) },
    ];
    this.bindListQueries();
    this.restoreState();
    this.loadLookups();
    this.onSearch(false);
  }

  private bindListQueries(): void {
    this.listQuery$.pipe(
      switchMap((query) => defer(() => {
        this.loadingList.set(true);
        this.loading.startLoading();
        this.errorMessage.set('');
        return this.service.getList(query).pipe(
          catchError((error) => {
            this.failList(apiErrorMessage(error, 'workOrder.loadError'));
            return EMPTY;
          }),
          finalize(() => {
            this.loadingList.set(false);
            this.loading.endLoading();
          }),
        );
      })), takeUntilDestroyed(this.destroyRef),
    ).subscribe((response) => {
      if (!response.isSuccess) {
        this.failList(response.message || 'workOrder.loadError');
        return;
      }
      this.data.set(response.entities ?? []);
      this.totalRecords.set(Number(response.totalCount ?? 0));
    });
  }

  private failList(message: string): void {
    this.errorMessage.set(message);
    this.data.set([]);
    this.totalRecords.set(0);
  }

  loadLookups(): void {
    if (this.loadingLookups()) return;
    this.loadingLookups.set(true);
    forkJoin({
      branches: this.branchService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
      customers: this.customerService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
      vehicles: this.vehicleService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
    }).pipe(takeUntilDestroyed(this.destroyRef), finalize(() => this.loadingLookups.set(false))).subscribe((response) => {
      if (response.branches?.isSuccess) this.branchOptions.set(response.branches.entities ?? []);
      if (response.customers?.isSuccess) this.customerOptions.set(response.customers.entities ?? []);
      if (response.vehicles?.isSuccess) this.vehicleOptions.set(response.vehicles.entities ?? []);
      this.lookupError.set(!response.branches?.isSuccess || !response.customers?.isSuccess || !response.vehicles?.isSuccess);
    });
  }

  onSearch(resetPage = true): void {
    if (this.printing() || this.reporting()) return;
    if (this.filterForm.invalid) {
      this.filterForm.markAllAsTouched();
      return;
    }
    const raw = this.filterForm.getRawValue();
    const [dateFrom, dateTo] = parseDateRange(raw.dateRange);
    this.activeFilters = {
      workOrderNo: raw.workOrderNo.trim(), customerPONo: raw.customerPONo.trim(),
      status: raw.status, acceptanceStatus: raw.acceptanceStatus, dateFrom, dateTo,
      vehicleId: raw.vehicleId, customerId: raw.customerId, branchId: raw.branchId, chassisNo: raw.chassisNo.trim(),
    };
    if (resetPage) this.pageNumber.set(1);
    this.requestList();
  }

  resetFilters(): void {
    this.filterForm.reset({ workOrderNo: '', customerPONo: '', status: StatusEnum.Open, acceptanceStatus: null,
      dateRange: currentMonthRange(), vehicleId: null, customerId: null, branchId: null, chassisNo: '' });
    this.onSearch();
  }

  onLazyLoad(event: TableLazyLoadEvent): void {
    if (this.printing() || this.reporting()) return;
    const paging = resolveListPaging(event, { pageNo: this.pageNumber(), pageSize: this.pageSize() });
    if (!paging.changed) return;
    this.pageNumber.set(paging.pageNo);
    this.pageSize.set(paging.pageSize);
    this.requestList();
  }

  private requestList(): void {
    writeListRoute(this.router, this.activatedRoute, this.returnQueryParams());
    this.listQuery$.next({ filters: { ...this.activeFilters }, pageNo: this.pageNumber(), pageSize: this.pageSize() });
  }

  private returnQueryParams(): Record<string, string | number | null | undefined> {
    return { ...this.activeFilters, status: this.activeFilters.status ?? 'all', dateRangeSet: 1, pageNo: this.pageNumber(), pageSize: this.pageSize() };
  }

  private restoreState(): void {
    const state = readListRoute(this.activatedRoute);
    const paging = state.paging();
    this.pageNumber.set(paging.pageNo);
    this.pageSize.set(paging.pageSize);
    const dates = state.number('dateRangeSet') === 1
      ? [state.text('dateFrom'), state.text('dateTo')].filter(Boolean).map((date) => new Date(`${parseDateValue(date)}T00:00:00`))
      : currentMonthRange();
    this.filterForm.patchValue({ workOrderNo: state.text('workOrderNo'), customerPONo: state.text('customerPONo'),
      status: state.text('status') === 'all' ? null : state.number('status') ?? StatusEnum.Open, acceptanceStatus: state.number('acceptanceStatus'), dateRange: dates.length ? dates : null,
      vehicleId: state.number('vehicleId'), customerId: state.number('customerId'), branchId: state.number('branchId'), chassisNo: state.text('chassisNo') });
  }

  openCreate(): void {
    if (this.actionBusy() || this.reporting() || this.printing()) return;
    void this.router.navigate(['/WorkOrder/create'], { queryParams: this.returnQueryParams() });
  }

  private openEditor(mode: 'view' | 'edit', row: WorkOrderListModel, print = false): void {
    if (this.actionBusy() || this.reporting() || this.printing()) return;
    void this.router.navigate(['/WorkOrder', mode, row.id], { queryParams: { ...this.returnQueryParams(), print: print ? 1 : null } });
  }

  private openCreateJob(row: WorkOrderListModel): void {
    if (this.actionBusy() || this.loadingList() || this.reporting() || this.printing()
      || Number(row.acceptanceStatus) !== AcceptanceStatusEnum.Accepted) return;
    this.actionBusy.set(true);
    void this.router.navigate(['/Job/create'], {
      queryParams: { ...this.returnQueryParams(), sourceWorkOrderId: row.id, returnTo: 'workOrder' },
    }).finally(() => this.actionBusy.set(false));
  }

  isPending(row: WorkOrderListModel): boolean {
    return Number(row.acceptanceStatus) === AcceptanceStatusEnum.NotAccepted;
  }

  acceptanceKey(value: number | null | undefined): string {
    return this.acceptanceStatusOptions.find((option) => option.id === Number(value))?.key ?? '';
  }

  private confirmAction(row: WorkOrderListModel, action: 'accept' | 'reject' | 'delete'): void {
    if (this.actionBusy() || this.reporting() || this.printing() || (action !== 'delete' && !this.isPending(row))) return;
    this.actionBusy.set(true);
    const record = { label: { key: 'workOrder.workOrderNo' }, title: row.no || String(row.id) };
    const confirmation = action === 'delete'
      ? this.confirmation.confirmDelete({ title: { key: 'workOrder.deleteConfirmTitle' }, message: { key: 'workOrder.deleteConfirmMessage' }, record })
      : this.confirmation.confirm({ severity: action === 'accept' ? 'info' : 'warning',
        title: { key: action === 'accept' ? 'workOrder.acceptConfirmTitle' : 'workOrder.rejectConfirmTitle' },
        message: { key: action === 'accept' ? 'workOrder.acceptConfirmMessage' : 'workOrder.rejectConfirmMessage' },
        confirmLabel: { key: action === 'accept' ? 'workOrder.accept' : 'workOrder.reject' }, record });
    confirmation.pipe(
      switchMap((confirmed) => {
        if (!confirmed) return EMPTY;
        this.loading.startLoading();
        this.errorMessage.set('');
        const request = action === 'accept' ? this.service.acceptWorkOrder(row.id)
          : action === 'reject' ? this.service.rejectWorkOrder(row.id) : this.service.deleteWorkOrder(row.id);
        return request.pipe(finalize(() => this.loading.endLoading()));
      }), takeUntilDestroyed(this.destroyRef), finalize(() => this.actionBusy.set(false)),
    ).subscribe({
      next: (response) => {
        if (!response.isSuccess) {
          this.errorMessage.set(response.message || 'workOrder.actionError');
          return;
        }
        if (action === 'delete' && this.data().length === 1 && this.pageNumber() > 1) this.pageNumber.update((page) => page - 1);
        this.requestList();
      }, error: (error) => this.errorMessage.set(apiErrorMessage(error, 'workOrder.actionError')),
    });
  }

  exportExcel(): void {
    if (this.exporting() || this.reporting() || this.printing()) return;
    this.exporting.set(true);
    this.service.getAllList({ ...this.activeFilters }).pipe(takeUntilDestroyed(this.destroyRef),
      finalize(() => this.exporting.set(false))).subscribe({
      next: (rows) => {
        if (!rows.length) {
          this.errorMessage.set('workOrder.empty');
          return;
        }
        onExportToExcel(signal(dataTableExportRows(rows, this.columns, (key) => this.translate.instant(key))), 'WorkOrders');
      }, error: (error) => this.errorMessage.set(error instanceof Error ? error.message : apiErrorMessage(error, 'workOrder.exportError')),
    });
  }

  onReport(): void {
    if (this.reporting() || this.printing() || this.actionBusy() || this.loadingList()) return;
    this.reporting.set(true);
    this.filterForm.disable({ emitEvent: false });
    this.loading.startLoading();
    this.service.getAllList({ ...this.activeFilters }).pipe(takeUntilDestroyed(this.destroyRef), finalize(() => {
      this.reporting.set(false);
      this.loading.endLoading();
      if (!this.printing()) this.filterForm.enable({ emitEvent: false });
    })).subscribe({
      next: (rows) => {
        if (!rows.length) {
          this.errorMessage.set('workOrder.empty');
          return;
        }
        this.printRows.set(rows);
        this.printSession.print({ afterPrint: () => {
          this.printRows.set([]);
          this.filterForm.enable({ emitEvent: false });
        } });
      }, error: (error) => this.errorMessage.set(error instanceof Error ? error.message : apiErrorMessage(error, 'workOrder.exportError')),
    });
  }
}

  ngOnInit(): void {
    this.mode.set(this.activatedRoute.snapshot.data['mode'] ?? 'create');
    const id = Number(this.activatedRoute.snapshot.paramMap.get('id'));
    this.recordId.set(Number.isInteger(id) && id > 0 ? id : null);
    if (!this.isReadOnly()) this.loadLookups();
    if (this.mode() === 'create') {
      this.itemsArray.push(this.createRow());
      this.workOrderForm.markAsPristine();

  loadLookups(): void {
    if (this.isReadOnly() || this.loadingLookups()) return;
    this.loadingLookups.set(true);
    forkJoin({
      customers: this.customerService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
      branches: this.branchService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
      vehicles: this.vehicleService.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }).pipe(catchError(() => of(null))),
      options: this.service.getOptions().pipe(catchError(() => of(null))),
    }).pipe(takeUntilDestroyed(this.destroyRef), finalize(() => this.loadingLookups.set(false))).subscribe((response) => {
      if (response.customers?.isSuccess) this.customerOptions.set(response.customers.entities ?? []);
      if (response.branches?.isSuccess) this.branchOptions.set(response.branches.entities ?? []);
      if (response.vehicles?.isSuccess) this.vehicleOptions.set(response.vehicles.entities ?? []);
      this.mergeSavedOptions();
      const rate = response.options?.entity?.effectiveTaxRatePercent;
      const optionsAvailable = response.options?.isSuccess === true && typeof rate === 'number' && Number.isFinite(rate);
      this.effectiveTaxRatePercent.set(optionsAvailable ? rate! : null);
      this.lookupError.set(!response.customers?.isSuccess || !response.branches?.isSuccess || !response.vehicles?.isSuccess || !optionsAvailable);
    });
  }

  loadDetails(): void {
    const id = this.recordId();
    if (this.loadingData() || this.saving()) return;
    if (!id) {
      this.errorMessage.set('workOrder.failedToLoadDetails');
      return;
    }
    this.loadingData.set(true);
    this.details.set(null);
    this.errorMessage.set('');
    this.workOrderForm.disable({ emitEvent: false });
    this.loading.startLoading();
    this.service.getDetails(id).pipe(takeUntilDestroyed(this.destroyRef), finalize(() => {
      this.loadingData.set(false);
      this.applyMode();
      this.loading.endLoading();
      if (this.details() && this.isReadOnly() && this.activatedRoute.snapshot.queryParamMap.get('print') === '1') this.print();
    })).subscribe({
      next: (response) => {
        if (!response.isSuccess || !response.entity) {
          this.errorMessage.set(response.message || 'workOrder.failedToLoadDetails');
          return;
        }
        this.details.set(response.entity);
        this.populateForm(response.entity);
        if (!this.isReadOnly() && !this.pending()) this.errorMessage.set('workOrder.editLocked');
      }, error: (error) => this.errorMessage.set(apiErrorMessage(error, 'workOrder.failedToLoadDetails')),
    });
  }

  private populateForm(dto: WorkOrderDetailsDTO): void {

    this.rows.replace(dto.workOrderItems ?? []);
    this.workOrderForm.markAsPristine();
    this.mergeSavedOptions();

  private mergeSavedOptions(): void {
    const dto = this.details();
    if (!dto) return;
    const choice = (id: number | null | undefined, value: string | null | undefined): DropDownSelect[] => id && value ? [{ id, value }] : [];
    this.customerOptions.set(mergeSelectOptions(this.customerOptions(), choice(dto.customerId, dto.customer?.name)));
    this.branchOptions.set(mergeSelectOptions(this.branchOptions(), choice(dto.branchId, dto.branch?.name)));
    const vehicles: DropDownSelect[] = [];
    for (const row of dto.workOrderItems) vehicles.push(...choice(row.vehicleId, row.vehicle?.plateNo || row.vehicle?.name));
    this.vehicleOptions.set(mergeSelectOptions(this.vehicleOptions(), vehicles));
  }

    if (this.isReadOnly()) return this.details()?.workOrderItems[index]?.tax ?? 0;
    const rate = this.effectiveTaxRatePercent();
    if (rate === null) return null;
    return allocateDocumentTax(this.itemsArray.controls.map(row => ({
      amount: roundMoney(Number(row.controls.amount.value || 0)), ratePercent: rate,

  }

  get summaryCards(): ReadonlyArray<SummaryCard> {
    const raw = this.workOrderForm.getRawValue();
    const amount = raw.workOrderItems.reduce((sum, row) => sum + roundMoney(Number(row.amount || 0)), 0);
    const tax = this.isReadOnly() ? raw.workOrderItems.reduce((sum, row) => sum + Number(row.tax || 0), 0)
      : this.itemsArray.controls.reduce((sum, _row, index) => sum + (this.taxPreview(index) ?? 0), 0);
    const gross = this.isReadOnly() ? this.details()?.totalAmount ?? 0
      : roundMoney(this.itemsArray.controls.reduce((sum, row, index) => sum + this.rowTotal(row, index), 0));
    const discount = roundMoney(Number(raw.discount || 0));
    const net = this.isReadOnly() ? this.details()?.netAmount ?? 0 : roundMoney(gross - discount);
    const previewKnown = this.isReadOnly() || this.effectiveTaxRatePercent() !== null;
    return [
      { id: 'amount', labelKey: 'workOrder.amount', value: amount, icon: 'bi bi-cash', accent: 'neutral', format: 'money' },
      { id: 'tax', labelKey: 'workOrder.tax', value: previewKnown ? tax : '—', icon: 'bi bi-receipt', accent: 'neutral', format: previewKnown ? 'money' : 'text' },
      { id: 'gross', labelKey: 'workOrder.amtPlusTax', value: previewKnown ? gross : '—', icon: 'bi bi-calculator', accent: 'info', format: previewKnown ? 'money' : 'text' },
      { id: 'discount', labelKey: 'workOrder.discount', value: discount, icon: 'bi bi-dash-circle', accent: 'neutral', format: 'money' },
      { id: 'net', labelKey: 'workOrder.netTotal', value: previewKnown ? net : '—', icon: 'bi bi-cash-stack', accent: 'primary', format: previewKnown ? 'money' : 'text' },
    ];
  }

  vehicleName(index: number): string {

  statusKey(): string {

  acceptanceKey(): string {

      : state === AcceptanceStatusEnum.Rejected ? 'workOrder.rejected' : 'workOrder.notAccepted';
  }

  save(andPrint = false): void {
    if (!this.canMutate() || this.loadingLookups() || this.lookupError()) return;
    if (this.workOrderForm.invalid || !this.itemsArray.length) {
      this.workOrderForm.markAllAsTouched();
      this.errorMessage.set('workOrder.validationError');
      this.focusFirstInvalid();
      return;
    }
    const raw = this.workOrderForm.getRawValue();
    const date = parseDateValue(raw.date);
    if (!date) return;
    const payload: WorkOrderCreateDTO = {
      date, customerPONo: raw.customerPONo.trim() || null, notes: raw.notes.trim() || null,
      customerId: raw.customerId!, branchId: raw.branchId!, discount: raw.discount,
      workOrderItems: this.itemsArray.controls.map((row): WorkOrderItemInputDTO => {
        const item = row.getRawValue();
        return { ...(this.mode() === 'edit' && this.rows.savedId(row) ? { id: this.rows.savedId(row) } : {}),
          description: item.description.trim(), vehicleId: item.vehicleId, amount: item.amount };
      }),
    };
    this.saving.set(true);
    this.workOrderForm.disable({ emitEvent: false });
    this.loading.startLoading();
    this.errorMessage.set('');
    const request = this.mode() === 'create' ? this.service.createWorkOrder(payload)
      : this.service.updateWorkOrder({ ...payload, id: this.recordId()! });
    request.pipe(takeUntilDestroyed(this.destroyRef), finalize(() => {
      this.saving.set(false);
      this.applyMode();
      this.loading.endLoading();
    })).subscribe({
      next: (response) => {
        if (!response.isSuccess) {
          this.errorMessage.set(response.message || 'workOrder.failedToSave');
          return;
        }
        this.saved = true;
        this.workOrderForm.markAsPristine();
        const id = Number(response.id ?? this.recordId());
        if (andPrint && Number.isInteger(id) && id > 0) {
          void this.router.navigate(['/WorkOrder/view', id], { queryParams: { ...this.returnQueryParams(), print: 1 } });
        } else {
          this.close();
        }
      }, error: (error) => this.errorMessage.set(apiErrorMessage(error, 'workOrder.failedToSave')),
    });
  }

  private applyMode(): void {
    if (this.canMutate()) this.workOrderForm.enable({ emitEvent: false });
    else this.workOrderForm.disable({ emitEvent: false });
  }

  private focusFirstInvalid(): void {
    const fields = [
      { control: this.workOrderForm.controls.date, id: 'wo-date' },
      { control: this.workOrderForm.controls.customerId, id: 'wo-customer' },
      { control: this.workOrderForm.controls.branchId, id: 'wo-branch' },
      { control: this.workOrderForm.controls.discount, id: 'wo-discount' },
    ];
    const master = fields.find((field) => field.control.invalid);
    if (master) {
      focusField(master.id);
      return;
    }
    const index = this.itemsArray.controls.findIndex((row) => row.invalid);
    if (index < 0) return;
    const row = this.itemsArray.at(index);
    const field = row.controls.description.invalid ? 'description' : row.controls.amount.invalid ? 'amount' : 'tax';
    focusField(`wo-${field}-${index}`);
  }

  canDeactivate(): boolean | Observable<boolean> {
    if (this.loadingData()) return false;
    if (this.saving() && !this.saved) return false;
    if (this.saved || this.isReadOnly() || this.workOrderForm.pristine) return true;
    return this.confirmation.confirmDiscard('workOrder');
  }

  @HostListener('window:beforeunload', ['$event'])
  beforeUnload(event: BeforeUnloadEvent): void {
    if (!this.saved && (this.saving() || (!this.isReadOnly() && this.workOrderForm.dirty))) {
      event.preventDefault();
      event.returnValue = '';
    }
  }

  requestClose(): void {
    if (!this.saving()) this.close();
  }

  private returnQueryParams(): Record<string, unknown> {
    const { print, ...params } = this.activatedRoute.snapshot.queryParams;
    return params;
  }

  private close(): void {
    void this.router.navigate(['/WorkOrder'], { queryParams: this.returnQueryParams() });
  }

  switchToEdit(): void {
    if (!this.isReadOnly() || !this.pending() || !this.recordId()) return;
    void this.router.navigate(['/WorkOrder/edit', this.recordId()], { queryParams: this.returnQueryParams() });
  }

  print(): void {
    if (this.details() && this.isReadOnly() && !this.loadingData()) this.printSession.print();
  }
}
