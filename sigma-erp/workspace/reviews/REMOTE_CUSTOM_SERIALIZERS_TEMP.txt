search(request: VehicleSearchModel, options?: { skipErrorInterceptor?: boolean }): Observable<Results<VehicleListModel>> {
    let params = new HttpParams();
    Object.entries(request).forEach(([key, value]) => {
      if (value === undefined || value === '') return;
      if (value === null) {
        // Vehicle/Search treats an empty pageSize as the confirmed unpaged contract.
        if (key === 'pageSize') params = params.set(key, '');
        return;
      }

      params = params.set(key, String(value));
    });
    return this.http.get<Results<VehicleListModel>>(`${environment.baseUrl}Vehicle/Search`, {
      params, ...(options?.skipErrorInterceptor ? { headers: { 'X-Skip-Error-Interceptor': 'true' } } : {}),
    });
  }

getList(query: FleetServiceLogListQuery): Observable<Results<FleetServiceLogListDTO>> {
    let params = new HttpParams()
      .set('PageNo', String(query.pageNo))
      .set('PageSize', String(query.pageSize))
      .set('SortField', query.sortField)
      .set('SortOrder', String(query.sortOrder));

    params = this.appendParam(params, 'PlateNo', query.plateNo);
    params = this.appendParam(params, 'DateFrom', query.dateFrom);
    params = this.appendParam(params, 'DateTo', query.dateTo);
    params = this.appendParam(
      params,
      'InternalWorkshopId',
      query.internalWorkshopId,
    );
    params = this.appendParam(params, 'ServiceItems', query.serviceItems);
    params = this.appendParam(params, 'WorkType', query.workType);

    return this.http.get<Results<FleetServiceLogListDTO>>(this.endpoint, { headers: { 'X-Skip-Error-Interceptor': 'true' },  params });
  }
private appendParam(
    params: HttpParams,
    key: string,
    value: string | number | null,
  ): HttpParams {
    return value === null || value === ''
      ? params
      : params.set(key, String(value));
  }
