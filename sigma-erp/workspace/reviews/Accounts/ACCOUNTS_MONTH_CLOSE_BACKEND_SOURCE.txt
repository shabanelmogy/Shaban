using System.Data;
using SiGma.ViewModels.ViewModels.MonthClose;

namespace SiGma.Business.Services.AccountServices;

public class MonthCloseService :
    Service<MonthCloseAddVM, MonthCloseUpdateVM, MonthCloseDetailVM, MonthCloseListVM, SiGmaDbContext, MonthClose>,
    IMonthCloseService
{
    public MonthCloseService(
        IUnitOfWork<SiGmaDbContext> unitOfWork,
        IMapper mapper,
        IStringLocalizer<GeneralMessage> localization)
        : base(unitOfWork, mapper, localization)
    {
    }

    /// <summary>
    /// Returns all months that fall within the fiscal year's date range.
    /// If month records don't exist yet, they are auto-created in the DB with Status = Open.
    /// Months already claimed by another fiscal year are included as-is (not duplicated).
    /// Safe to call multiple times — duplicate months are skipped gracefully.
    /// </summary>
    public async Task<Result<List<MonthCloseListVM>>> GetByFiscalYearAsync(int fiscalYearId, CancellationToken cancellationToken = default)
    {
        var fiscalYear = await UnitOfWork.Repository.FindFirstByAsync<FiscalYear>(
            x => x.Id == fiscalYearId);

        if (fiscalYear is null)
            return new Result<List<MonthCloseListVM>> { IsSuccess = false, Message = "Fiscal year not found." };

        var fiscalYearName = fiscalYear.StartDate.ToString("dd/MM/yyyy") + " to " + fiscalYear.EndDate.ToString("dd/MM/yyyy");

        // Build the full list of calendar months in this fiscal year's range
        var allMonths = new List<(int Year, int Month)>();
        var current = new DateTime(fiscalYear.StartDate.Year, fiscalYear.StartDate.Month, 1);
        var end = new DateTime(fiscalYear.EndDate.Year, fiscalYear.EndDate.Month, 1);
        while (current <= end)
        {
            allMonths.Add((current.Year, current.Month));
            current = current.AddMonths(1);
        }

        // Load records already saved for this fiscal year
        var existing = await UnitOfWork.Repository
            .Query<MonthClose>(x => x.FiscalYearId == fiscalYearId)
            .AsNoTracking()
            .ToListAsync(cancellationToken);

        var existingKeys = existing.Select(x => (x.Year, x.Month)).ToHashSet();

        // Also check which months are already claimed by OTHER fiscal years (global unique constraint)
        var years = allMonths.Select(m => m.Year).Distinct().ToList();
        var months = allMonths.Select(m => m.Month).Distinct().ToList();

        var claimedElsewhere = await UnitOfWork.Repository
            .Query<MonthClose>(x => x.FiscalYearId != fiscalYearId
                                 && years.Contains(x.Year)
                                 && months.Contains(x.Month))
            .AsNoTracking()
            .Select(x => new { x.Year, x.Month })
            .ToListAsync(cancellationToken);

        var claimedKeys = claimedElsewhere.Select(x => (x.Year, x.Month)).ToHashSet();

        // Only insert months not yet saved anywhere
        var newRecords = allMonths
            .Where(m => !existingKeys.Contains(m) && !claimedKeys.Contains(m))
            .Select(m => new MonthClose
            {
                FiscalYearId = fiscalYearId,
                Month = m.Month,
                Year = m.Year,
                Status = MonthCloseStatus.Open
            })
            .ToList();

        if (newRecords.Count > 0)
        {
            try
            {
                await UnitOfWork.Repository.AddRangeAsync(newRecords, cancellationToken);
                await UnitOfWork.SaveChangesAsync(cancellationToken);
            }
            catch (DbUpdateException)
            {
                // Concurrent request already inserted — clear tracker and re-load
                UnitOfWork.Context.ChangeTracker.Clear();
                existing = await UnitOfWork.Repository
                    .Query<MonthClose>(x => x.FiscalYearId == fiscalYearId)
                    .AsNoTracking()
                    .ToListAsync(cancellationToken);
                newRecords.Clear();
            }
        }

        // Merge existing + newly inserted, sort by calendar order
        var allRecords = existing.Concat(newRecords).ToList();

        var result = allRecords
            .OrderBy(x => x.Year).ThenBy(x => x.Month)
            .Select(x =>
            {
                var vm = Mapper.Map<MonthCloseListVM>(x);
                vm.FiscalYearName = fiscalYearName;
                return vm;
            })
            .ToList();

        return new Result<List<MonthCloseListVM>> { IsSuccess = true, Entity = result };
    }

    public override async Task<Result> AddAsync(MonthCloseAddVM vm, CancellationToken cancellationToken)
    {
        if (vm.Status != MonthCloseStatus.Open)
            return LegacyResult(WrapCheck(Check([Issue("useCloseAction")])));

        if (vm.Month is < 1 or > 12 || vm.Year is < 1900 or > 2100)
            return LegacyResult(WrapCheck(Check([Issue("invalidPeriod")])));

        var invalidInput = await InvalidInputAsync(vm, id: null, cancellationToken: cancellationToken);
        if (invalidInput is not null) return invalidInput;

        var periodIssues = await EligibilityIssuesAsync(new MonthClose
        {
            FiscalYearId = vm.FiscalYearId, Month = vm.Month, Year = vm.Year, Status = vm.Status
        }, closing: false, cancellationToken);
        if (periodIssues.Count > 0) return LegacyResult(WrapCheck(Check(periodIssues)));

        // Validate fiscal year exists
        if (!await UnitOfWork.Repository.ExistsAsync<FiscalYear>(x => x.Id == vm.FiscalYearId, cancellationToken: cancellationToken))
            return new Result { IsSuccess = false, Message = "Fiscal year not found." };

        // Prevent duplicate month within the same fiscal year
        var exists = await UnitOfWork.Repository.ExistsAsync<MonthClose>(x =>
            x.FiscalYearId == vm.FiscalYearId &&
            x.Year == vm.Year &&
            x.Month == vm.Month,
            cancellationToken: cancellationToken);

        if (exists)
            return new Result { IsSuccess = false, Message = "A month close record already exists for this month and fiscal year." };

        var entity = Mapper.Map<MonthClose>(vm);

        await UnitOfWork.Repository.AddAsync(entity, cancellationToken);
        await UnitOfWork.SaveChangesAsync(cancellationToken);

        return new Result { IsSuccess = true, Id = entity.Id, Message = "Month close record added successfully." };
    }

    public override async Task<Result> UpdateAsync(MonthCloseUpdateVM vm)
    {
        // Period identity is immutable; status writes use the same close/reopen guards.
        var result = await ChangeStatusAsync(vm.Id, vm.Status, CancellationToken.None, vm);
        return LegacyResult(result);
    }

    public override async Task<Result> RemoveAsync(int id)
    {
        await using var transaction = await UnitOfWork.Context.Database.BeginTransactionAsync(IsolationLevel.Serializable);
        var entity = await UnitOfWork.Repository.Query<MonthClose>(x => x.Id == id).FirstOrDefaultAsync();
        if (entity is null) return LegacyResult(WrapCheck(Check([Issue("monthNotFound")])));
        if (entity.Status != MonthCloseStatus.Open)
            return LegacyResult(WrapCheck(Check([Issue("closedRecordDelete")])));

        var result = await base.RemoveAsync(id);
        if (result.IsSuccess) await transaction.CommitAsync();
        return result;
    }

    public async Task<Result> ToggleStatusAsync(int id, CancellationToken cancellationToken = default)
        => LegacyResult(await ChangeStatusAsync(id, null, cancellationToken));

    public Task<Result<MonthCloseCheckVM>> CloseMonthAsync(int id, CancellationToken cancellationToken = default)
        => ChangeStatusAsync(id, MonthCloseStatus.Closed, cancellationToken);

    public Task<Result<MonthCloseCheckVM>> ReopenMonthAsync(int id, CancellationToken cancellationToken = default)
        => ChangeStatusAsync(id, MonthCloseStatus.Open, cancellationToken);

    public Task<Result<MonthCloseCheckVM>> VerifyValidMonthAsync(int id, CancellationToken cancellationToken = default)
        => ReadCheckAsync(id, 1, cancellationToken);

    public Task<Result<MonthCloseCheckVM>> VerifyVoucherApprovalsAsync(int id, CancellationToken cancellationToken = default)
        => ReadCheckAsync(id, 2, cancellationToken);

    public Task<Result<MonthCloseCheckVM>> VerifyDataInconsistencyAsync(int id, CancellationToken cancellationToken = default)
        => ReadCheckAsync(id, 3, cancellationToken);

    private async Task<Result<MonthCloseCheckVM>> ReadCheckAsync(int id, int step, CancellationToken cancellationToken)
    {
        var entity = await UnitOfWork.Repository.Query<MonthClose>(x => x.Id == id)
            .AsNoTracking().FirstOrDefaultAsync(cancellationToken);
        if (entity is null) return WrapCheck(Check([Issue("monthNotFound")]));

        // Every read stage requires a still-eligible period. Final close repeats all checks atomically.
        var issues = await EligibilityIssuesAsync(entity, closing: true, cancellationToken);
        if (issues.Count == 0 && step == 2)
            issues.AddRange(await VoucherApprovalIssuesAsync(entity, cancellationToken));
        if (issues.Count == 0 && step == 3)
            issues.AddRange(await VoucherIntegrityIssuesAsync(entity, cancellationToken));
        return WrapCheck(Check(issues));
    }

    private async Task<Result<MonthCloseCheckVM>> ChangeStatusAsync(
        int id, MonthCloseStatus? desiredStatus, CancellationToken cancellationToken, MonthCloseUpdateVM? update = null)
    {
        // Mirrors FiscalYear close: eligibility reads and the state write commit together.
        await using var transaction = await UnitOfWork.Context.Database.BeginTransactionAsync(
            IsolationLevel.Serializable, cancellationToken);
        var entity = await UnitOfWork.Repository.Query<MonthClose>(x => x.Id == id)
            .FirstOrDefaultAsync(cancellationToken);
        if (entity is null) return WrapCheck(Check([Issue("monthNotFound")]));
        if (entity.Status is not (MonthCloseStatus.Open or MonthCloseStatus.Closed))
            return WrapCheck(Check([Issue("invalidStatus")]));
        if (update is not null && (update.FiscalYearId != entity.FiscalYearId ||
            update.Month != entity.Month || update.Year != entity.Year))
            return WrapCheck(Check([Issue("periodImmutable")]));

        var target = desiredStatus ?? (entity.Status == MonthCloseStatus.Open
            ? MonthCloseStatus.Closed : MonthCloseStatus.Open);
        if (target is not (MonthCloseStatus.Open or MonthCloseStatus.Closed))
            return WrapCheck(Check([Issue("invalidStatus")]));
        if (entity.Status == target)
            return WrapCheck(Check([], target == MonthCloseStatus.Closed ? "MonthCloseAlreadyClosed" : "MonthCloseAlreadyOpen"), id);

        var closing = target == MonthCloseStatus.Closed;
        var issues = await EligibilityIssuesAsync(entity, closing, cancellationToken);
        if (closing && IsCalendarPeriodValid(entity))
        {
            // Gather every financial blocker even when a separate eligibility check fails.
            issues.AddRange(await VoucherApprovalIssuesAsync(entity, cancellationToken));
            issues.AddRange(await VoucherIntegrityIssuesAsync(entity, cancellationToken));
        }
        if (issues.Count > 0) return WrapCheck(Check(issues));

        if (update is not null)
        {
            var invalidInput = await InvalidInputAsync(update, id);
            if (invalidInput is not null)
                return new Result<MonthCloseCheckVM> { IsSuccess = false, Message = invalidInput.Message };
            Mapper.Map(update, entity);
        }
        entity.Status = target;
        UnitOfWork.Repository.Update(entity);
        await UnitOfWork.SaveChangesAsync(cancellationToken);
        await transaction.CommitAsync(cancellationToken);
        return WrapCheck(Check([], closing ? "MonthCloseClosed" : "MonthCloseReopened"), id);
    }

    private async Task<List<MonthCloseIssueVM>> EligibilityIssuesAsync(
        MonthClose entity, bool closing, CancellationToken cancellationToken)
    {
        var issues = new List<MonthCloseIssueVM>();
        if (!IsCalendarPeriodValid(entity)) return [Issue("invalidPeriod")];
        if (entity.Status is not (MonthCloseStatus.Open or MonthCloseStatus.Closed))
            return [Issue("invalidStatus")];

        var fiscal = await UnitOfWork.Repository.Query<FiscalYear>(x => x.Id == entity.FiscalYearId)
            .AsNoTracking().FirstOrDefaultAsync(cancellationToken);
        if (fiscal is null) return [Issue("fiscalYearNotFound")];
        var start = new DateTime(entity.Year, entity.Month, 1);
        var end = start.AddMonths(1);
        if (fiscal.StartDate > fiscal.EndDate || fiscal.StartDate.Date >= end || fiscal.EndDate.Date < start)
            return [Issue("invalidPeriod")];
        if (fiscal.Status != FiscalYearStatus.Open) issues.Add(Issue("fiscalYearClosed"));
        if (!closing) return issues;
        if (entity.Status != MonthCloseStatus.Open) issues.Add(Issue("monthAlreadyClosed"));

        var currentMonth = new DateTime(DateTime.Today.Year, DateTime.Today.Month, 1);
        if (start > currentMonth) issues.Add(Issue("futureMonth"));
        var fiscalStartMonth = new DateTime(fiscal.StartDate.Year, fiscal.StartDate.Month, 1);
        var expectedPrevious = (start.Year - fiscalStartMonth.Year) * 12 + start.Month - fiscalStartMonth.Month;
        if (expectedPrevious > 0)
        {
            var startKey = fiscalStartMonth.Year * 100 + fiscalStartMonth.Month;
            var targetKey = entity.Year * 100 + entity.Month;
            var closedPrevious = await UnitOfWork.Repository.Query<MonthClose>(x =>
                    x.FiscalYearId == entity.FiscalYearId && x.Status == MonthCloseStatus.Closed &&
                    x.Month >= 1 && x.Month <= 12 &&
                    x.Year * 100 + x.Month >= startKey && x.Year * 100 + x.Month < targetKey)
                .AsNoTracking().CountAsync(cancellationToken);
            if (closedPrevious < expectedPrevious)
                issues.Add(Issue("previousMonthsOpen", expectedPrevious - closedPrevious));
        }
        return issues;
    }

    private async Task<List<MonthCloseIssueVM>> VoucherApprovalIssuesAsync(MonthClose entity, CancellationToken cancellationToken)
    {
        var start = new DateTime(entity.Year, entity.Month, 1);
        var end = start.AddMonths(1);
        var count = await UnitOfWork.Repository.Query<JournalVoucher>(x =>
                x.Date >= start && x.Date < end && !x.IsVoided &&
                (x.ApprovalStatus != ApprovalStatus.Approved || !x.IsPosted))
            .AsNoTracking().CountAsync(cancellationToken);
        return count > 0 ? [Issue("vouchersNotApproved", count)] : [];
    }

    private async Task<List<MonthCloseIssueVM>> VoucherIntegrityIssuesAsync(MonthClose entity, CancellationToken cancellationToken)
    {
        var start = new DateTime(entity.Year, entity.Month, 1);
        var end = start.AddMonths(1);
        var fiscal = await UnitOfWork.Repository.Query<FiscalYear>(x => x.Id == entity.FiscalYearId)
            .AsNoTracking().FirstOrDefaultAsync(cancellationToken);
        if (fiscal is null) return [];
        var firstDate = fiscal.StartDate.Date;
        var lastDate = fiscal.EndDate.Date.AddDays(1);
        var count = await UnitOfWork.Repository.Query<JournalVoucher>(x =>
                x.Date >= start && x.Date < end && !x.IsVoided)
            .AsNoTracking()
            .Select(x => new
            {
                WrongPeriod = x.FiscalYearId != entity.FiscalYearId || x.Date < firstDate || x.Date >= lastDate,
                HasDetails = x.JournalVoucherDetails.Any(),
                HasInvalidLine = x.JournalVoucherDetails.Any(detail =>
                    !detail.AccountId.HasValue ||
                    (detail.Debit ?? 0m) < 0m || (detail.Credit ?? 0m) < 0m ||
                    ((detail.Debit ?? 0m) > 0m && (detail.Credit ?? 0m) > 0m) ||
                    ((detail.Debit ?? 0m) == 0m && (detail.Credit ?? 0m) == 0m)),
                TotalDebit = x.JournalVoucherDetails.Sum(detail => detail.Debit ?? 0m),
                TotalCredit = x.JournalVoucherDetails.Sum(detail => detail.Credit ?? 0m)
            })
            .CountAsync(x => x.WrongPeriod || !x.HasDetails || x.HasInvalidLine || x.TotalDebit != x.TotalCredit, cancellationToken);
        return count > 0 ? [Issue("invalidVouchers", count)] : [];
    }

    private static bool IsCalendarPeriodValid(MonthClose entity)
        => entity.Month is >= 1 and <= 12 && entity.Year is >= 1900 and <= 2100;

    private static MonthCloseIssueVM Issue(string code, int count = 1) => new() { Code = code, Count = count };

    private MonthCloseCheckVM Check(List<MonthCloseIssueVM> issues, string successKey = "MonthCloseCheckPassed")
        => new()
        {
            IsValid = issues.Count == 0,
            Message = Localization.GetString(issues.Count == 0 ? successKey : "MonthCloseBlocked"),
            IssueCount = issues.Sum(x => x.Count),
            Issues = issues
        };

    private static Result<MonthCloseCheckVM> WrapCheck(MonthCloseCheckVM check, int? id = null)
        => new() { IsSuccess = true, Entity = check, Id = id, Message = check.Message };

    private Result LegacyResult(Result<MonthCloseCheckVM> result)
        => new()
        {
            IsSuccess = result.IsSuccess && result.Entity?.IsValid == true,
            Message = result.Message,
            Id = result.Id,
            Errors = result.Entity?.Issues.Select(issue =>
                Localization.GetString("MonthCloseReason_" + issue.Code, issue.Count).Value).ToList() ?? []
        };
}
