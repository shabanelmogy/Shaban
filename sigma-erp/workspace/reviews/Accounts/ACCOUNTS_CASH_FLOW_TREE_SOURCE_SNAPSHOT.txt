import { Component, inject } from '@angular/core';
import { TranslateService } from '@ngx-translate/core';

import { CashFlowDTO, CashFlowFilter, CashFlowLine } from '../../models/list';
import { CashFlowStatementRow } from '../../models/cash-flow-statement-row';
import { CashFlowStatementService } from '../../services/cash-flow-statement.service';
import { AccountReportPageComponent } from '../../../shared/report-family/account-report-page.component';
import { AccountReportConfig } from '../../../shared/report-family/account-report.models';
import { reportNumber, reportDate } from '../../../shared/report-family/account-report-values';

import { DropDownSelect } from 'src/app/modules/shared/models/dropdown-select';

import { DataTableColumn, dataTableExportRows } from 'src/app/modules/shared/components/data-table/data-table.component';

@Component({
  selector: 'app-accounts', standalone: true,
  imports: [AccountReportPageComponent],
  templateUrl: './list.component.html', styleUrl: './list.component.scss',
  host: { class: 'sigma-route-host sigma-route-host--full-width sigma-print-flow' },
})
export class CashFlowStatementComponent {
  private readonly service = inject(CashFlowStatementService);
  private readonly fiscalYears = inject(FiscalYearService);
  private readonly router = inject(Router);
  private readonly translate = inject(TranslateService);
  private readonly columns: ReadonlyArray<DataTableColumn<CashFlowStatementRow>> = [
    {
      field: 'name', header: 'cashFlowStatement.account', width: '78%',
      cellClass: 'sigma-data-table__cell--wrap',
      value: row => row.labelKey
        ? this.translate.instant(row.labelKey)
        : [row.code, row.name].filter(Boolean).join(' — ') || '—',
    },
    {
      field: 'amount', header: 'cashFlowStatement.amount', width: '22%', type: 'money',
      headerClass: 'text-end', cellClass: 'text-end sigma-data-table__cell--numeric',
    },
  ];
  readonly config: AccountReportConfig<CashFlowDTO, CashFlowFilter, CashFlowStatementRow> = {
    id: 'cashFlowStatement', titleKey: 'cashFlowStatement.title', icon: 'bi bi-file-earmark-spreadsheet', exportFileName: 'CashFlowStatement',
    filters: [
      { key: 'fiscalYearId', labelKey: 'cashFlowStatement.fiscalYear', kind: 'select', initialValue: null, requiredLookup: true, loadOptions: () => this.fiscalYears.getSelectList<Results<DropDownSelect>>({ skipErrorInterceptor: true }) },
      { key: 'asOfDate', labelKey: 'cashFlowStatement.asOfDate', kind: 'date', initialValue: null }
    ],
    load: filter => this.service.getReport(filter),
    buildFilter: raw => {

      return { filter: { fiscalYearId: reportNumber(raw, 'fiscalYearId'), asOfDate: reportDate(raw, 'asOfDate') } };
    },
    sections: response => [
      {
        id: 'statement', dataKey: 'id', minWidth: '32rem',
        rows: this.statementRows(response), columns: this.columns, rowTone: row => row.tone,
        hierarchy: {
          rowKey: row => row.id,
          children: row => row.children,
          defaultExpanded: true,
          activateEnabled: row => row.accountId > 0 && row.tone === null,
          activate: row => this.openLedger(row.accountId),
          activateTitle: 'cashFlowStatement.openLedger',
        },
      },
    ],
    exportRows: (response, _sections, translate) => this.exportRows(response, translate),
  };

  private statementRows(response: CashFlowDTO): CashFlowStatementRow[] {
    const rows: CashFlowStatementRow[] = [];
    const activities = [
      { key: 'operatingActivities', lines: response.operatingActivities ?? [], totalKey: 'netCashFromOperating', total: response.netCashFromOperating },
      { key: 'investingActivities', lines: response.investingActivities ?? [], totalKey: 'netCashFromInvesting', total: response.netCashFromInvesting },
      { key: 'financingActivities', lines: response.financingActivities ?? [], totalKey: 'netCashFromFinancing', total: response.netCashFromFinancing },
    ];
    for (const activity of activities) {
      const children: CashFlowStatementRow[] = [];
      if (activity.key === 'operatingActivities' && !activity.lines.some(line => this.isNetIncomeLine(line))) {
        children.push(this.statementLabel('netIncome', response.netIncome, null));
      }
      activity.lines.forEach((line, index) => children.push({
        ...line, id: `${activity.key}-${index}`, tone: null,
        labelKey: activity.key === 'operatingActivities' && this.isNetIncomeLine(line)
          ? 'cashFlowStatement.netIncome' : undefined,
      }));
      children.push(this.statementLabel(activity.totalKey, activity.total, 'subtotal'));
      rows.push({ ...this.statementLabel(activity.key, activity.total, 'section'), children });
    }
    rows.push(this.statementLabel('netChangeInCash', response.netChangeInCash, 'total'));
    return rows;
  }

  private statementLabel(key: string, amount: number | null, tone: CashFlowStatementRow['tone']): CashFlowStatementRow {
    return { id: key, accountId: 0, code: '', name: '', labelKey: `cashFlowStatement.${key}`, amount, tone };
  }

  private isNetIncomeLine(line: CashFlowLine): boolean {
    return line.accountId === 0 && !line.code && line.name === 'Net Income';
  }

  private exportRows(response: CashFlowDTO, translate: (key: string) => string): Array<Record<string, string | number>> {
    const columns: ReadonlyArray<DataTableColumn<CashFlowLine>> = [
      { field: 'name', header: 'cashFlowStatement.account', value: row => [row.code, row.name].filter(Boolean).join(' — ') || '—' },
      { field: 'amount', header: 'cashFlowStatement.amount', type: 'money' },
    ];
    const activities = [
      { key: 'operatingActivities', lines: response.operatingActivities ?? [] },
      { key: 'investingActivities', lines: response.investingActivities ?? [] },
      { key: 'financingActivities', lines: response.financingActivities ?? [] },
    ];
    const rows: Array<Record<string, string | number>> = [];
    for (const activity of activities) {
      for (const row of dataTableExportRows(activity.lines, columns, translate)) {
        rows.push({
          [translate('accountReports.section')]: translate(`cashFlowStatement.${activity.key}`), ...row,
        });
      }
    }
    return rows;
  }

  private openLedger(accountId: number): void {
    if (accountId > 0) void this.router.navigate(['/Ledger'], { queryParams: { accountId } });
  }
}

export class DataTableComponent<T extends object> implements OnChanges {
  private readonly translate = inject(TranslateService);
  @ViewChild(Table) private readonly table?: Table;

  /** Reset the shared row viewport without exposing PrimeNG internals to features. */

  get rows(): number {
    return this._rows;
  }
  set rows(value: number | null | undefined) {
    if (!value) {
      this._rows = DEFAULT_PAGE_SIZE;
      return;
    }
    // 10 stays the minimum page size (the token interceptor enforces it on requests too).
    this._rows = Math.max(10, value);
  }

  private _rowsPerPageOptions: number[] = [10, 15, 20, 50, 100];

  @Input()
  get rowsPerPageOptions(): number[] {
    return this._rowsPerPageOptions;
  }
  set rowsPerPageOptions(options: number[] | null | undefined) {
    if (!options || options.length === 0 || this.isLegacyRowsPerPage(options)) {
      this._rowsPerPageOptions = [10, 15, 20, 50, 100];
      return;
    }
    this._rowsPerPageOptions = options;
  }

  private isLegacyRowsPerPage(options: number[]): boolean {
    const key = options.join(',');
    return (
      key === '10,20,50,100' ||
      key === '5,10,25,50' ||
      key === '5,10,20,50,100' ||
      key === '10,25,50' ||
      key === '5,10,25' ||
      options.some((opt) => opt < 10)
    );
  }

  /**
   * Fill mode (UI block 6, backlog 34): the grid takes the remaining height of its flex parent,
   * rows scroll inside it and the paginator stays at the bottom. Replaces the feature-level
   * `::ng-deep` fill workaround; `scrollHeight` is ignored.

  @Input() fill = false;
  @Input() showCurrentPageReport?: boolean;

  @Output() readonly selectionChange = new EventEmitter<T[]>();

  private readonly initializedHierarchyKeys = new Set<string>();
  private readonly hierarchyLevels = new Map<string, number>();

    if (this.fill) {

    }
    if (!this.scrollHeight || this.scrollHeight === '100%') {
      return '288px';
    }
    return this.scrollHeight;
  }

  protected get resolvedFirst(): number {
    if (this.first !== undefined && this.first !== null) {
      return this.first;
    }
    return Math.max(0, (this.page - 1) * this.rows);
  }

  protected get resolvedTotalRecords(): number {
    if (!this.lazy && (!this.totalRecords || this.totalRecords === 0)) {
      return this.value?.length ?? 0;
    }
    return this.totalRecords;
  }

  protected get renderedValue(): T[] {
    return this.hierarchy ? this.visibleHierarchyRows : this.value;
  }

  protected get resolvedShowCurrentPageReport(): boolean {
    if (this.showCurrentPageReport !== undefined && this.showCurrentPageReport !== null) {
      return this.showCurrentPageReport;
    }
    return !!this.currentPageReport || (this.paginator && this.resolvedTotalRecords > 0);
  }

  protected get resolvedCurrentPageReport(): string {
    return this.currentPageReport || '{first} - {last} / {totalRecords}';
  }

  protected onTableLazyLoad(event: TableLazyLoadEvent): void {
    const rawRows = Number(event.rows ?? this.rows);
    const newRows = Math.max(10, Number.isNaN(rawRows) || rawRows < 10 ? 10 : rawRows);
    const newFirst = Math.max(0, Number(event.first ?? 0));
    const newPage = Math.floor(newFirst / newRows) + 1;

    this.first = newFirst;

    if (this._rows !== newRows) {
      this._rows = newRows;
      this.rowsChange.emit(newRows);
    }

    if (this.page !== newPage) {
      this.page = newPage;
      this.pageChange.emit(newPage);
    }

    const sortField = typeof event.sortField === 'string'
      ? event.sortField
      : (Array.isArray(event.sortField) ? event.sortField[0] : undefined);
    const sortOrder = event.sortOrder === 1 || event.sortOrder === -1
      ? event.sortOrder
      : undefined;

    this.paginationChange.emit({
      page: newPage,
      pageSize: newRows,
      first: newFirst,
      sortField,
      sortOrder,
    });

    this.lazyLoad.emit({
      ...event,
      rows: newRows,
      first: newFirst,
    });
  }

  /** PrimeNG local sorting resets first without emitting onPage. Keep controlled paging in sync. */
  protected onTableFirstChange(first: number): void {
    if (this.lazy || !this.paginator || first === this.resolvedFirst) return;
    this.onTablePage({ first, rows: this.rows });
  }

  protected onTablePage(event: { first: number; rows: number; page?: number }): void {
    const newRows = Math.max(1, Number(event.rows ?? this.rows));
    const newFirst = Math.max(0, Number(event.first ?? 0));
    const newPage = event.page !== undefined ? event.page + 1 : Math.floor(newFirst / newRows) + 1;

    this.first = newFirst;

    if (this.rows !== newRows) {
      this.rows = newRows;
      this.rowsChange.emit(newRows);
    }

    if (this.page !== newPage) {
      this.page = newPage;
      this.pageChange.emit(newPage);
    }

    this.paginationChange.emit({
      page: newPage,
      pageSize: newRows,
      first: newFirst,
    });
  }

  protected setActionTemplate(template: TemplateRef<unknown>): void {
    this.actionTemplate = template;
  }

  protected displayValue(row: T, column: DataTableColumn<T>): DataTableCellValue {
    return dataTableCellText(row, column, (key) => this.translate.instant(key));
  }

  protected statusOf(row: T, column: DataTableColumn<T>): DataTableStatus | null {
    return column.type === 'status' && column.status ? column.status(row) : null;
  }

  protected statusClass(status: DataTableStatus): string {
    return !status.tone || status.tone === 'neutral'
      ? 'sigma-status-badge'
      : `sigma-status-badge sigma-status-badge--${status.tone}`;
  }

  protected selectionLabel(row: T): string {
    return this.rowSelectionAriaLabel?.(row) ?? '';
  }

  protected hierarchyLevel(row: T): number {
    const hierarchy = this.hierarchy;
    return hierarchy ? this.hierarchyLevels.get(String(hierarchy.rowKey(row))) ?? 0 : 0;
  }

  protected columnIndent(row: T, column: DataTableColumn<T>): number {
    return Math.max(0, Math.floor(column.indent?.(row) ?? 0));
  }

  protected hasHierarchyChildren(row: T): boolean {
    return (this.hierarchy?.children(row)?.length ?? 0) > 0;
  }

  protected hierarchyExpanded(row: T): boolean {
    const hierarchy = this.hierarchy;
    return !!hierarchy && this.expandedHierarchyKeys.has(String(hierarchy.rowKey(row)));
  }

  protected toggleHierarchy(row: T, event: Event): void {
    event.stopPropagation();
    const hierarchy = this.hierarchy;
    if (!hierarchy || !this.hasHierarchyChildren(row)) return;
    const key = String(hierarchy.rowKey(row));
    if (this.expandedHierarchyKeys.has(key)) this.expandedHierarchyKeys.delete(key);
    else this.expandedHierarchyKeys.add(key);
    this.rebuildHierarchyRows();
  }

  protected hierarchyCode(row: T): string {
    return String(this.hierarchy?.code?.(row) ?? '');
  }

  protected hierarchyLabel(row: T, column: DataTableColumn<T>): string {
    return String(this.hierarchy?.label?.(row) ?? this.displayValue(row, column) ?? '');
  }

  protected hierarchyActivate(row: T, event: Event): void {
    event.stopPropagation();
    const hierarchy = this.hierarchy;
    if (!hierarchy?.activateEnabled || hierarchy.activateEnabled(row)) hierarchy?.activate?.(row);
  }

  protected hierarchyActivateAriaLabel(row: T, column: DataTableColumn<T>): string {
    return this.hierarchy?.activateAriaLabel?.(row) || this.hierarchyLabel(row, column);
  }

  protected cellLinkActivate(row: T, column: DataTableColumn<T>, event: Event): void {
    event.stopPropagation();
    if (!column.link?.enabled || column.link.enabled(row)) column.link?.activate(row);
  }

  protected cellLinkAriaLabel(row: T, column: DataTableColumn<T>): string {
    return String(column.link?.ariaLabel?.(row) || this.displayValue(row, column) || '');
  }

  protected updateSelection(value: T[] | T | null): void {
    if (this.selectionDisabled) return;
    this.selectionChange.emit(Array.isArray(value) ? value : []);
  }

  protected activateRow(row: T, event: Event): void {
    if (!this.hasRowActivation) return;

    const target = event.target;
    if (target instanceof Element) {
      const interactiveTarget = target.closest(
        'button, a[href], input, select, textarea, option, label, summary, '
          + '[contenteditable="true"], [tabindex]:not([tabindex="-1"]), [role]',
      );
      if (interactiveTarget && interactiveTarget !== event.currentTarget) return;
    }

    if (event.type === 'keydown') event.preventDefault();
    this.rowActivated.emit(row);
  }

  private rebuildHierarchyRows(): void {
    this.hierarchyLevels.clear();
    if (!this.hierarchy) {
      this.visibleHierarchyRows = [];
      return;
    }
    const rows: T[] = [];
    this.appendHierarchyRows(this.value, 0, rows, this.hierarchy);
    this.visibleHierarchyRows = rows;
  }

  private appendHierarchyRows(
    source: ReadonlyArray<T>,
    level: number,
    target: T[],
    hierarchy: DataTableHierarchy<T>,
  ): void {
    source.forEach((row) => {
      const key = String(hierarchy.rowKey(row));
      if (!this.initializedHierarchyKeys.has(key)) {
        this.initializedHierarchyKeys.add(key);
        if (hierarchy.defaultExpanded) this.expandedHierarchyKeys.add(key);
      }
      this.hierarchyLevels.set(key, level);
      target.push(row);
      const children = hierarchy.children(row) ?? [];
      if (children.length && (this.printAllRows || this.expandedHierarchyKeys.has(key))) {
        this.appendHierarchyRows(children, level + 1, target, hierarchy);
      }
    });
  }
}

            (keydown)="$event.stopPropagation()"
          >
            <p-tableCheckbox
              [disabled]="selectionDisabled"
              [value]="row"
              [ariaLabel]="selectionLabel(row)"
            ></p-tableCheckbox>
          </td>
        }
        @if (hasActions) {
          <td class="sigma-data-table__actions-column">
            @if (actionTemplate) {
              <ng-container
                *ngTemplateOutlet="actionTemplate; context: { $implicit: row }"
              ></ng-container>
            }
          </td>
        }
        @for (column of columns; track column.field + ':' + column.header; let isFirst = $first) {
          <td [ngClass]="column.cellClass" [style.width]="column.width || null">
            @if (column.cellTemplate) {
              <ng-container
                *ngTemplateOutlet="column.cellTemplate; context: {
                  $implicit: row,
                  row,
                  column
                }"
              ></ng-container>
            } @else if (hierarchy && isFirst) {
              <div
                class="sigma-data-table__hierarchy-cell"
                [style.padding-inline-start.px]="hierarchyLevel(row) * 14"
              >
                @if (hasHierarchyChildren(row)) {
                  <button
                    type="button"
                    class="sigma-data-table__hierarchy-toggle"
                    [attr.aria-expanded]="hierarchyExpanded(row)"
                    [attr.aria-label]="hierarchyLabel(row, column)"
                    (click)="toggleHierarchy(row, $event)"
                  >
                    <i
                      class="bi"
                      [class.bi-chevron-down]="hierarchyExpanded(row)"
                      [class.bi-chevron-right]="!hierarchyExpanded(row)"
                      aria-hidden="true"
                    ></i>
                  </button>
                } @else {
                  <span class="sigma-data-table__hierarchy-spacer" aria-hidden="true"></span>
                }
                @if (hierarchyCode(row); as code) {
                  <span class="sigma-data-table__hierarchy-code">{{ code }}</span>
                }
                @if (hierarchy.activate && (!hierarchy.activateEnabled || hierarchy.activateEnabled(row))) {
                  <button
                    type="button"
                    class="sigma-data-table__cell-link"
                    [title]="hierarchy.activateTitle ? (hierarchy.activateTitle | translate) : ''"
                    [attr.aria-label]="hierarchyActivateAriaLabel(row, column)"
                    (click)="hierarchyActivate(row, $event)"
                  >
                    {{ hierarchyLabel(row, column) }}
                  </button>
                } @else {
                  <span>{{ hierarchyLabel(row, column) }}</span>
                }
              </div>
            } @else if (column.link && (!column.link.enabled || column.link.enabled(row))) {
              <div
                class="sigma-data-table__indented-cell"
                [style.padding-inline-start.px]="columnIndent(row, column) * 14"
              >
                <button
                  type="button"
                  class="sigma-data-table__cell-link"
                  [title]="column.link.title ? (column.link.title | translate) : ''"
                  [attr.aria-label]="cellLinkAriaLabel(row, column)"
                  (click)="cellLinkActivate(row, column, $event)"
                >
                  {{ displayValue(row, column) }}
                </button>
              </div>
            } @else if (column.indent) {
              <div
                class="sigma-data-table__indented-cell"
                [style.padding-inline-start.px]="columnIndent(row, column) * 14"
              >
                {{ displayValue(row, column) }}
              </div>
            } @else {
              @if (statusOf(row, column); as status) {
                <span [class]="statusClass(status)">{{ status.label | translate }}</span>
              } @else {
                {{ displayValue(row, column) }}
              }
            }
          </td>
        }
      </tr>
    </ng-template>
