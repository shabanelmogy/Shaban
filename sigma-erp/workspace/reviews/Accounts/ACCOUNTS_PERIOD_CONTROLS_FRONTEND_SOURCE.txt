import { CommonModule } from '@angular/common';
import { Component, DestroyRef, EventEmitter, Input, OnInit, Output, inject, signal } from '@angular/core';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { TranslateModule } from '@ngx-translate/core';
import { Observable, finalize, timeout } from 'rxjs';
import { EditorDialogComponent } from 'src/app/modules/shared/components/editor-dialog/editor-dialog.component';

type CloseStep = 1 | 2 | 3 | 4;
const CLOSE_CHECK_TIMEOUT_MS = 20_000;

@Component({
  selector: 'app-month-close',
  standalone: true,
  imports: [CommonModule, TranslateModule, EditorDialogComponent, PrimaryActionButtonComponent, StateMessageComponent, StepFormComponent, MonthCloseCheckResultComponent],
  templateUrl: './close.component.html',
  styleUrl: './close.component.scss',
})
export class MonthCloseComponent extends BaseComponentService implements OnInit {
  @Input({ required: true }) record!: MonthCloseDTO;
  @Input({ required: true }) monthLabel!: string;

  @Output() readonly cancelled = new EventEmitter<void>();
  @Output() readonly completed = new EventEmitter<string>();

  private readonly monthCloseService = inject(MonthCloseService);
  private readonly confirmationDialog = inject(ConfirmationDialogService);
  private readonly destroyRef = inject(DestroyRef);

  readonly navigationSteps: ReadonlyArray<StepFormStep> = [
    { key: '1', label: 'monthClose.checks.steps.validate' },
    { key: '2', label: 'monthClose.checks.steps.approvals' },
    { key: '3', label: 'monthClose.checks.steps.consistency' },
    { key: '4', label: 'monthClose.checks.steps.close' },
  ];
  readonly currentStep = signal<CloseStep>(1);
  readonly results = signal<Partial<Record<CloseStep, MonthCloseCheckDTO>>>({});
  readonly processing = signal(false);
  readonly confirmationPending = signal(false);
  readonly closeSucceeded = signal(false);
  readonly visible = signal(true);

  ngOnInit(): void {
    this.runCurrentStep();
  }

  currentResult(): MonthCloseCheckDTO | null {
    return this.results()[this.currentStep()] ?? null;
  }

  canContinue(): boolean {
    return !this.processing() && !this.confirmationPending() && this.currentResult()?.isValid === true;
  }

  next(): void {
    if (!this.canContinue() || this.currentStep() >= 3) return;
    this.currentStep.update(step => (step + 1) as CloseStep);
    if (!this.currentResult()) this.runCurrentStep();
  }

  previous(): void {
    if (this.processing() || this.confirmationPending() || this.currentStep() <= 1 || this.currentStep() >= 4) return;
    this.currentStep.update(step => (step - 1) as CloseStep);
  }

  requestCloseMonth(): void {
    if (!this.canContinue() || this.currentStep() !== 3) return;
    const id = this.record.id;
    this.confirmationPending.set(true);
    this.confirmationDialog.confirm({
      severity: 'danger',
      title: { key: 'monthClose.confirm.closeTitle' },
      message: { key: 'monthClose.confirm.closeMessage', params: { record: `${this.monthLabel} ${this.record.year}` } },
      record: {
        label: { key: 'monthClose.confirm.recordLabel', params: { month: this.monthLabel, year: this.record.year } },
        title: `${this.monthLabel} ${this.record.year}`,
        icon: 'bi bi-calendar-month',
      },
      confirmLabel: { key: 'monthClose.actions.close' },
      cancelLabel: { key: 'general.cancel' },
      confirmIcon: 'bi bi-lock-fill',
      icon: 'bi bi-lock-fill',
    }).pipe(
      takeUntilDestroyed(this.destroyRef),
      finalize(() => this.confirmationPending.set(false)),
    ).subscribe(confirmed => {
      if (!confirmed || !this.visible() || this.record.id !== id || this.processing() || this.currentStep() !== 3 || this.currentResult()?.isValid !== true) return;
      this.currentStep.set(4);
      this.runCurrentStep();
    });
  }

  retry(): void {
    if (!this.processing() && !this.confirmationPending() && !this.closeSucceeded()) this.runCurrentStep();
  }

  close(): void {
    if (this.processing() || this.confirmationPending()) return;
    this.visible.set(false);
    this.cancelled.emit();
  }

  onEscape(event: Event): void {
    event.preventDefault();
    event.stopPropagation();
    this.close();
  }

  private runCurrentStep(): void {
    if (!this.record.id || this.processing()) return;
    const step = this.currentStep();
    const id = this.record.id;
    this.processing.set(true);
    this.loading.startLoading();
    const request = this.getStepRequest(step);
    (step === 4 ? request : request.pipe(timeout({ first: CLOSE_CHECK_TIMEOUT_MS }))).pipe(
      takeUntilDestroyed(this.destroyRef),
      finalize(() => { this.processing.set(false); this.loading.endLoading(); }),
    ).subscribe({
      next: response => {
        if (!this.visible() || this.record.id !== id) return;
        const result: MonthCloseCheckDTO = response.isSuccess && response.entity
          ? response.entity
          : {
              isValid: false,
              message: response.message || 'monthClose.checks.failed',
              issueCount: response.entity?.issueCount ?? 0,
              issues: response.entity?.issues ?? [],
            };
        this.results.update(results => ({ ...results, [step]: result }));
        if (step === 4 && response.isSuccess && result.isValid && !this.closeSucceeded()) {
          this.closeSucceeded.set(true);
          this.completed.emit(response.message || result.message || 'general.success');
        }
      },
      error: (error: unknown) => {
        if (!this.visible() || this.record.id !== id) return;
        this.results.update(results => ({
          ...results,
          [step]: { isValid: false, message: apiErrorMessage(error, 'monthClose.checks.failed'), issueCount: 0, issues: [] },
        }));
      },
    });
  }

  private getStepRequest(step: CloseStep): Observable<Result<MonthCloseCheckDTO>> {
    switch (step) {
      case 1: return this.monthCloseService.verifyValidMonth(this.record.id);
      case 2: return this.monthCloseService.verifyVoucherApprovals(this.record.id);
      case 3: return this.monthCloseService.verifyDataInconsistency(this.record.id);
      case 4: return this.monthCloseService.closeMonth(this.record.id);
    }
  }
}
